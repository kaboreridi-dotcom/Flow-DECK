   189→  return names.map((n, i) => createColumn(boardId, n, i));

   192→function seedWelcomeBoard(lang: Lang): AppData {
   193→  const t = (k: string) => messages[lang][k] ?? k;
   194→  const board = createBoard(t("seed.board.name"), t("seed.board.desc"), 0);
   195→  const cols = defaultColumnsFor(board.id, [t("col.todo"), t("col.inprogress"), t("col.done")]);
   196→  const welcome = createTask({
   197→    boardId: board.id,
   198→    columnId: cols[0].id,
   199→    title: t("seed.task.title"),
   200→    description: t("seed.task.desc"),
   201→    priority: "normal",
   202→  });
   203→  const done1 = createTask({ boardId: board.id, columnId: cols[2].id, title: t("seed.task2.title") });
   204→  done1.status = "done";
   205→  done1.completedAt = Date.now();
   206→  return {
   207→    ...emptyData(),
   208→    boards: [board],
   209→    columns: cols,
   210→    tasks: [welcome, done1],
   211→    activeBoardId: board.id,
   212→  };
   213→}

   215→function nextOrder(tasks: Task[], columnId: string): number {
   216→  return tasks.filter((t) => t.columnId === columnId).reduce((m, t) => Math.max(m, t.order + 1), 0);
   217→}

   219→export function buildBoardExport(data: AppData, boardId: string) {
   220→  const board = data.boards.find((b) => b.id === boardId);
   221→  if (!board) return null;
   222→  const columns = data.columns.filter((c) => c.boardId === boardId);
   223→  const columnIds = new Set(columns.map((c) => c.id));
   224→  const tasks = data.tasks.filter((t) => t.boardId === boardId && columnIds.has(t.columnId));
   225→  return {
   226→    app: "Flow DECK",
   227→    type: "board",
   228→    version: data.version,
   229→    exportedAt: new Date().toISOString(),
   230→    boards: [board],
   231→    columns,
   232→    tasks,
   233→  };
   234→}

   248→export const useAppStore = create<AppStore>((set, get) => {
   249→  const pushHistory = () => {
   250→    const s = get();
   251→    if (s._batching) return;
   252→    const snap = snapshotOf(s.data);
   253→    set({ past: [...s.past.slice(-MAX_HISTORY + 1), snap], future: [] });
   254→  };
   255→
   256→  const applyData = (mutator: (d: AppData) => AppData, touchBoards = false) => {
   257→    const s = get();
   258→    const now = Date.now();
   259→    let d = mutator(s.data);
   260→    if (touchBoards && d.activeBoardId) {
   261→      d = {
   262→        ...d,
   263→        boards: d.boards.map((b) => (b.id === d.activeBoardId ? { ...b, updatedAt: now } : b)),
   264→      };
   265→    }
   266→    set({ data: d });
   267→    saveData(d);
   268→  };
   269→
   270→  const withTask = (id: string, mutator: (t: Task) => Task, touch = true) => {
   271→    applyData((d) => {
   272→      const now = Date.now();
   273→      return {
   274→        ...d,
   275→        tasks: d.tasks.map((t) => (t.id === id ? { ...mutator(t), updatedAt: touch ? now : t.updatedAt } : t)),
   276→      };
   277→    }, touch);
   278→  };
   279→
   280→  return {
   281→    data: emptyData(),
   282→    ui: {
   283→      view: "kanban",
   284→      search: "",
   285→      globalSearchOpen: false,
   286→      sortKey: "manual",
   287→      sortDir: "asc",
   288→      filters: EMPTY_FILTERS,
   289→      openTaskId: null,
   290→      sidebarOpen: false,
   291→      statsOpen: false,
   292→      archiveOpen: false,
   293→      trashOpen: false,
   294→      templatesOpen: false,
   295→      importExportOpen: false,
   296→      shortcutsOpen: false,
   297→      focusOpen: false,
   298→      newTaskOpen: false,
   299→      settingsOpen: false,
   300→    },
   301→    hydrated: false,
   302→    past: [],
   303→    future: [],
   304→    _batching: false,
   305→    _batchSnapshot: null,
   306→
   307→    hydrate: () => {
   308→      if (get().hydrated) return;
   309→      const { data, corrupted } = loadData();
   310→      let d: AppData;
   311→      if (data) {
   312→        d = data;
   313→      } else {
   314→        const lang: Lang = detectBrowserLang();
   315→        d = seedWelcomeBoard(lang);
   316→        saveData(d, true);
   317→      }
   318→      set({
   319→        data: d,
   320→        hydrated: true,
   321→        ui: {
   322→          ...get().ui,
   323→          view: d.prefs?.view ?? "kanban",
   324→          sortKey: d.prefs?.sortKey ?? "manual",
   325→          sortDir: d.prefs?.sortDir ?? "asc",
   326→        },
   327→      });
   328→      if (corrupted) {
   329→        window.dispatchEvent(new CustomEvent("flowdeck:corrupted"));
   330→      }
   331→    },
   332→
   333→    setUI: (patch) => {
   334→      set({ ui: { ...get().ui, ...patch } });
   335→      const ui = get().ui;
   336→      if (patch.view || patch.sortKey || patch.sortDir) {
   337→        const d = { ...get().data, prefs: { view: ui.view, sortKey: ui.sortKey, sortDir: ui.sortDir } };
   338→        set({ data: d });
   339→        saveData(d);
   340→      }
   341→    },
   342→    setSearch: (q) => set({ ui: { ...get().ui, search: q } }),
   343→    setView: (v) => get().setUI({ view: v }),
   344→    setSort: (key, dir) =>
   345→      get().setUI({ sortKey: key, sortDir: dir ?? get().ui.sortDir }),
   346→    toggleSortDir: () => get().setUI({ sortDir: get().ui.sortDir === "asc" ? "desc" : "asc" }),
   347→    setFilters: (patch) => set({ ui: { ...get().ui, filters: { ...get().ui.filters, ...patch } } }),
   348→    resetFilters: () => set({ ui: { ...get().ui, filters: { ...EMPTY_FILTERS } } }),
   349→
   350→    // ---------- boards ----------
   351→    createBoard: (opts) => {
   352→      const lang: Lang = detectBrowserLang();
   353→      const t = (k: string) => messages[lang][k] ?? k;
   354→      const name = opts?.name?.trim() || t("board.new");
   355→      const board = createBoard(name, opts?.description ?? "", get().data.boards.length);
   356→      const cols = defaultColumnsFor(board.id, [t("col.todo"), t("col.inprogress"), t("col.done")]);
   357→      pushHistory();
   358→      applyData((d) => ({
   359→        ...d,
   360→        boards: [...d.boards, board],
   361→        columns: [...d.columns, ...cols],
   362→        activeBoardId: board.id,
   363→      }));
   364→      return board.id;
   365→    },
   366→
   367→    createBoardWithStructure: (spec) => {
   368→      const board = createBoard(spec.name, spec.description ?? "", get().data.boards.length);
   369→      const cols = spec.columns.map((n, i) => createColumn(board.id, n, i));
   370→      const labels: Label[] = (spec.labels ?? []).map((l) => createLabel(l.name, l.color));
   371→      board.labels = labels;
   372→      const orderCounter = new Map<string, number>();
   373→      const tasks: Task[] = (spec.tasks ?? []).map((spec2) => {
   374→        const col = cols[Math.min(Math.max(0, spec2.columnIndex), cols.length - 1)];
   375→        const order = orderCounter.get(col.id) ?? 0;
   376→        orderCounter.set(col.id, order + 1);
   377→        return createTask({
   378→          boardId: board.id,
   379→          columnId: col.id,
   380→          title: spec2.title,
   381→          description: spec2.description,
   382→          priority: spec2.priority,
   383→          order,
   384→        });
   385→      });
   386→      pushHistory();
   387→      applyData((d) => ({
   388→        ...d,
   389→        boards: [...d.boards, board],
   390→        columns: [...d.columns, ...cols],
   391→        tasks: [...d.tasks, ...tasks],
   392→        activeBoardId: board.id,
   393→      }));
   394→      return board.id;
   395→    },
   396→
   397→    updateBoard: (id, patch) => {
   398→      pushHistory();
   399→      applyData((d) => ({
   400→        ...d,
   401→        boards: d.boards.map((b) => (b.id === id ? { ...b, ...patch, updatedAt: Date.now() } : b)),
   402→      }));
   403→    },
   404→
   405→    duplicateBoard: (id) => {
   406→      const s = get();
   407→      const src = s.data.boards.find((b) => b.id === id);
   408→      if (!src) return;
   409→      const copy = { ...src, id: uid("b"), name: `${src.name} (2)`, createdAt: Date.now(), updatedAt: Date.now() };
   410→      const colMap = new Map<string, string>();
   411→      const cols: Column[] = s.data.columns
   412→        .filter((c) => c.boardId === id)
   413→        .map((c) => {
   414→          const nid = uid("c");
   415→          colMap.set(c.id, nid);
   416→          return { ...c, id: nid, boardId: copy.id, createdAt: Date.now() };
   417→        });
   418→      const tasks = s.data.tasks
   419→        .filter((t) => t.boardId === id)
   420→        .map((t) => {
   421→          const c = cloneTask(t);
   422→          c.id = uid("t");
   423→          c.boardId = copy.id;
   424→          c.columnId = colMap.get(t.columnId) ?? cols[0]?.id ?? "";
   425→          return c;
   426→        });
   427→      pushHistory();
   428→      applyData((d) => ({
   429→        ...d,
   430→        boards: [...d.boards, copy],
   431→        columns: [...d.columns, ...cols],
   432→        tasks: [...d.tasks, ...tasks],
   433→        activeBoardId: copy.id,
   434→      }));
   435→    },
   436→
   437→    deleteBoard: (id) => {
   438→      const s = get();
   439→      const board = s.data.boards.find((b) => b.id === id);
   440→      if (!board) return;
   441→      const cols = s.data.columns.filter((c) => c.boardId === id);
   442→      const tasks = s.data.tasks.filter((t) => t.boardId === id);
   443→      const item: TrashItem = {
   444→        id: uid("x"),
   445→        type: "board",
   446→        deletedAt: Date.now(),
   447→        label: board.name,
   448→        boardId: board.id,
   449→        payloadBoard: board,
   450→        payloadColumns: cols,
   451→        payloadTasks: tasks,
   452→      };
   453→      pushHistory();
   454→      applyData((d) => {
   455→        const remaining = d.boards.filter((b) => b.id !== id);
   456→        const nextActive =
   457→          d.activeBoardId === id ? remaining.find((b) => !b.archived)?.id ?? null : d.activeBoardId;
   458→        return {
   459→          ...d,
   460→          boards: remaining,
   461→          columns: d.columns.filter((c) => c.boardId !== id),
   462→          tasks: d.tasks.filter((t) => t.boardId !== id),
   463→          trash: [...d.trash, item],
   464→          activeBoardId: nextActive,
   465→        };
   466→      });
   467→    },
   468→
   469→    resetBoard: (id) => {
   470→      const s = get();
   471→      const board = s.data.boards.find((b) => b.id === id);
   472→      if (!board) return;
   473→      const cols = s.data.columns.filter((c) => c.boardId === id);
   474→      const tasks = s.data.tasks.filter((t) => t.boardId === id);
   475→      const lang: Lang = detectBrowserLang();
   476→      const t = (k: string) => messages[lang][k] ?? k;
   477→      const items: TrashItem[] = [
   478→        ...tasks.map((task) => ({
   479→          id: uid("x"),
   480→          type: "task" as const,
   481→          deletedAt: Date.now(),
   482→          label: task.title,
   483→          boardId: id,
   484→          columnId: task.columnId,
   485→          payloadTask: task,
   486→        })),
   487→        ...cols.map((c) => ({
   488→          id: uid("x"),
   489→          type: "column" as const,
   490→          deletedAt: Date.now(),
   491→          label: c.name,
   492→          boardId: id,
   493→          columnId: c.id,
   494→          payloadColumn: c,
   495→          payloadTasks: [],
   496→        })),
   497→      ];
   498→      const fresh = defaultColumnsFor(id, [t("col.todo"), t("col.inprogress"), t("col.done")]);
   499→      pushHistory();
   500→      applyData((d) => ({
   501→        ...d,
   502→        columns: [...d.columns.filter((c) => c.boardId !== id), ...fresh],
   503→        tasks: d.tasks.filter((t2) => t2.boardId !== id),
   504→        trash: [...d.trash, ...items],
   505→      }));
   506→    },
   507→
   508→    setActiveBoard: (id) => {
   509→      applyData((d) => ({ ...d, activeBoardId: id }), false);
   510→      get().setSearch("");
   511→    },
   512→
   513→    moveBoardTo: (id, toIndex) => {
   514→      pushHistory();
   515→      applyData((d) => {
   516→        const sorted = [...d.boards].sort((a, b) => a.order - b.order);
   517→        const from = sorted.findIndex((b) => b.id === id);
   518→        if (from < 0) return d;
   519→        const clamped = Math.max(0, Math.min(toIndex, sorted.length - 1));
   520→        if (clamped === from) return d;
   521→        const [moved] = sorted.splice(from, 1);
   522→        sorted.splice(clamped, 0, moved);
   523→        const orderMap = new Map(sorted.map((b, i) => [b.id, i]));
   524→        return { ...d, boards: d.boards.map((b) => ({ ...b, order: orderMap.get(b.id) ?? b.order })) };
   525→      });
   526→    },
   527→
   528→    archiveBoard: (id) => {
   529→      pushHistory();
   530→      applyData((d) => {
   531→        const boards = d.boards.map((b) => (b.id === id ? { ...b, archived: true } : b));
   532→        const nextActive =
   533→          d.activeBoardId === id
   534→            ? boards.find((b) => b.id !== id && !b.archived)?.id ?? null
   535→            : d.activeBoardId;
   536→        return { ...d, boards, activeBoardId: nextActive };
   537→      });
   538→    },
   539→
   540→    unarchiveBoard: (id) => {
   541→      pushHistory();
   542→      applyData((d) => ({
   543→        ...d,
   544→        boards: d.boards.map((b) => (b.id === id ? { ...b, archived: false } : b)),
   545→      }));
   546→    },
   547→
   548→    // ---------- columns ----------
   549→    addColumn: (boardId, name) => {
   550→      const order = get().data.columns.filter((c) => c.boardId === boardId).length;
   551→      const col = createColumn(boardId, name.trim() || "?", order);
   552→      pushHistory();
   553→      applyData((d) => ({ ...d, columns: [...d.columns, col] }), true);
   554→    },
   555→
   556→    updateColumn: (id, patch) => {
   557→      pushHistory();
   558→      applyData((d) => ({
   559→        ...d,
   560→        columns: d.columns.map((c) => (c.id === id ? { ...c, ...patch } : c)),
   561→      }));
   562→    },
   563→
   564→    deleteColumn: (id, mode, targetColumnId) => {
   565→      const s = get();
   566→      const col = s.data.columns.find((c) => c.id === id);
   567→      if (!col) return;

  1057→        future: [current, ...s.future.slice(0, MAX_HISTORY - 1)],
  1058→      });
  1059→      saveData(get().data, true);
  1060→    },
  1061→    redo: () => {
  1062→      const s = get();
  1063→      if (s.future.length === 0) return;
  1064→      const next = s.future[0];
  1065→      const current = snapshotOf(s.data);
  1066→      set({
  1067→        data: { ...s.data, ...next },
  1068→        past: [...s.past.slice(-MAX_HISTORY + 1), current],
  1069→        future: s.future.slice(1),
  1070→      });
  1071→      saveData(get().data, true);
  1072→    },
  1073→    beginBatch: () => {
  1074→      set({ _batching: true, _batchSnapshot: snapshotOf(get().data) });
  1075→    },
  1076→    commitBatch: () => {
  1077→      const s = get();
  1078→      if (s._batchSnapshot) {
  1079→        set({
  1080→          past: [...s.past.slice(-MAX_HISTORY + 1), s._batchSnapshot],
  1081→          future: [],
  1082→          _batching: false,
  1083→          _batchSnapshot: null,
  1084→        });
  1085→      } else {
  1086→        set({ _batching: false, _batchSnapshot: null });
  1087→      }
  1088→      saveData(get().data, true);
  1089→    },
  1090→    cancelBatch: () => {
  1091→      const s = get();
  1092→      if (s._batchSnapshot) {
  1093→        set({ data: { ...s.data, ...s._batchSnapshot }, _batching: false, _batchSnapshot: null });
  1094→      } else {
  1095→        set({ _batching: false, _batchSnapshot: null });
  1096→      }
  1097→    },
  1098→
  1099→    // ---------- import ----------
  1100→    importBoards: (payload, mode) => {
  1101→      const summary: ImportSummary = { added: 0, replaced: 0, skipped: 0 };
  1102→      pushHistory();
  1103→      applyData((d) => {
  1104→        let nd = { ...d };
  1105→        for (const src of payload.boards) {
  1106→          const existing = nd.boards.find((b) => b.id === src.id);
  1107→          if (existing && mode === "replace") {
  1108→            const cols = payload.columns.filter((c) => c.boardId === src.id);
  1109→            const colIds = new Set(cols.map((c) => c.id));
  1110→            const tasks = payload.tasks.filter((t) => t.boardId === src.id && colIds.has(t.columnId));
  1111→            nd = {
  1112→              ...nd,
  1113→              boards: nd.boards.map((b) => (b.id === src.id ? { ...src, updatedAt: Date.now() } : b)),
  1114→              columns: [...nd.columns.filter((c) => c.boardId !== src.id), ...cols],
  1115→              tasks: [...nd.tasks.filter((t) => t.boardId !== src.id || !colIds.has(t.columnId)), ...tasks],
  1116→            };
  1117→            summary.replaced += 1;
  1118→          } else if (existing && mode === "copy") {
  1119→            // Full copy with fresh ids.
  1120→            const boardIdMap = new Map<string, string>();
  1121→            const nb: Board = { ...src, id: uid("b"), name: `${src.name} (2)`, createdAt: Date.now(), updatedAt: Date.now() };
  1122→            boardIdMap.set(src.id, nb.id);
  1123→            const labelMap = new Map<string, string>();
  1124→            const labels: Label[] = (src.labels ?? []).map((l) => {
  1125→              const nid = uid("l");
  1126→              labelMap.set(l.id, nid);
  1127→              return { ...l, id: nid };
  1128→            });
  1129→            nb.labels = labels;
  1130→            const colMap = new Map<string, string>();
  1131→            const cols: Column[] = payload.columns
  1132→              .filter((c) => c.boardId === src.id)
  1133→              .map((c) => {
  1134→                const nid = uid("c");
  1135→                colMap.set(c.id, nid);
  1136→                return { ...c, id: nid, boardId: nb.id, createdAt: Date.now() };
  1137→              });
  1138→            const tasks: Task[] = payload.tasks
  1139→              .filter((t) => t.boardId === src.id && colMap.has(t.columnId))
  1140→              .map((t) => {
  1141→                const c = cloneTask(t);
  1142→                c.id = uid("t");
  1143→                c.boardId = nb.id;
  1144→                c.columnId = colMap.get(t.columnId) ?? cols[0]?.id ?? "";
  1145→                c.labelIds = t.labelIds.map((lid) => labelMap.get(lid) ?? lid).filter((lid) => labels.some((l) => l.id === lid));
  1146→                return c;
  1147→              });
  1148→            nd = {
  1149→              ...nd,
  1150→              boards: [...nd.boards, nb],
  1151→              columns: [...nd.columns, ...cols],
  1152→              tasks: [...nd.tasks, ...tasks],
  1153→              activeBoardId: nb.id,
  1154→            };
  1155→            summary.added += 1;
  1156→          } else {
  1157→            const cols = payload.columns.filter((c) => c.boardId === src.id);
  1158→            const colIds = new Set(cols.map((c) => c.id));
  1159→            const tasks = payload.tasks.filter((t) => t.boardId === src.id && colIds.has(t.columnId));
  1160→            nd = {
  1161→              ...nd,
  1162→              boards: [...nd.boards, { ...src, order: nd.boards.length }],
  1163→              columns: [...nd.columns, ...cols],
  1164→              tasks: [...nd.tasks, ...tasks],
  1165→              activeBoardId: src.id,
  1166→            };
  1167→            summary.added += 1;
  1168→          }
  1169→        }
  1170→        return nd;
  1171→      });
  1172→      return summary;
  1173→    },
  1174→
  1175→    replaceAllData: (data) => {
  1176→      const clean = sanitizeAppData(data) ?? emptyData();
  1177→      set({ past: [], future: [], data: clean });
  1178→      saveData(clean, true);
  1179→    },
  1180→  };
  1181→});
